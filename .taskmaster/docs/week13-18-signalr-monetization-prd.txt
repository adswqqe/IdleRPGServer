## Feature 14: Real-time Chat System (SignalR) (Week 13)

### 게임 메커니즘
- 전체 채팅
- 길드 채팅
- 귓속말 (1:1)
- 채팅 금지 시스템

### 기술 요구사항

**SignalR Hub**:
```csharp
public class ChatHub : Hub {
    private readonly IUserService _userService;

    // 전체 채팅
    public async Task SendGlobalMessage(string message) {
        var user = await GetCurrentUser();
        if (user.IsMuted) {
            await Clients.Caller.SendAsync("Error", "채팅 금지 상태입니다");
            return;
        }

        await Clients.All.SendAsync("ReceiveGlobalMessage", user.Username, message, DateTime.Now);
    }

    // 길드 채팅
    public async Task SendGuildMessage(int guildId, string message) {
        var user = await GetCurrentUser();
        var member = await _guildService.GetGuildMember(user.Id, guildId);
        if (member == null) {
            await Clients.Caller.SendAsync("Error", "길드 멤버만 채팅 가능합니다");
            return;
        }

        await Clients.Group($"Guild_{guildId}").SendAsync("ReceiveGuildMessage", user.Username, message, DateTime.Now);
    }

    // 귓속말
    public async Task SendWhisper(string targetUsername, string message) {
        var user = await GetCurrentUser();
        var targetConnectionId = await GetConnectionId(targetUsername);

        if (targetConnectionId == null) {
            await Clients.Caller.SendAsync("Error", "사용자를 찾을 수 없습니다");
            return;
        }

        await Clients.Client(targetConnectionId).SendAsync("ReceiveWhisper", user.Username, message, DateTime.Now);
        await Clients.Caller.SendAsync("WhisperSent", targetUsername, message, DateTime.Now);
    }

    // 길드 그룹 가입
    public async Task JoinGuildChannel(int guildId) {
        await Groups.AddToGroupAsync(Context.ConnectionId, $"Guild_{guildId}");
    }

    // 길드 그룹 탈퇴
    public async Task LeaveGuildChannel(int guildId) {
        await Groups.RemoveFromGroupAsync(Context.ConnectionId, $"Guild_{guildId}");
    }

    // Connection 관리
    public override async Task OnConnectedAsync() {
        var user = await GetCurrentUser();
        _connectionManager.AddConnection(user.Id, Context.ConnectionId);
        await base.OnConnectedAsync();
    }

    public override async Task OnDisconnectedAsync(Exception exception) {
        var user = await GetCurrentUser();
        _connectionManager.RemoveConnection(user.Id);
        await base.OnDisconnectedAsync(exception);
    }
}
```

**Connection Manager**:
```csharp
public class ConnectionManager {
    private readonly Dictionary<int, string> _connections = new();

    public void AddConnection(int userId, string connectionId) {
        _connections[userId] = connectionId;
    }

    public void RemoveConnection(int userId) {
        _connections.Remove(userId);
    }

    public string GetConnectionId(int userId) {
        return _connections.TryGetValue(userId, out var connectionId) ? connectionId : null;
    }
}
```

```csharp
using Microsoft.AspNetCore.SignalR.Client;

public class ChatManager : MonoBehaviour {
    private HubConnection _connection;

    public async Task ConnectToChat() {
        _connection = new HubConnectionBuilder()
            .WithUrl("https://localhost:7122/chatHub", options => {
                options.AccessTokenProvider = () => Task.FromResult(NetworkManager.Instance.GetAccessToken());
            })
            .Build();

        // 이벤트 수신
        _connection.On<string, string, DateTime>("ReceiveGlobalMessage", (username, message, timestamp) => {
            UnityMainThreadDispatcher.Instance().Enqueue(() => {
                OnGlobalMessageReceived?.Invoke(username, message, timestamp);
            });
        });

        _connection.On<string, string, DateTime>("ReceiveGuildMessage", (username, message, timestamp) => {
            UnityMainThreadDispatcher.Instance().Enqueue(() => {
                OnGuildMessageReceived?.Invoke(username, message, timestamp);
            });
        });

        _connection.On<string, string, DateTime>("ReceiveWhisper", (username, message, timestamp) => {
            UnityMainThreadDispatcher.Instance().Enqueue(() => {
                OnWhisperReceived?.Invoke(username, message, timestamp);
            });
        });

        await _connection.StartAsync();
    }

    public async Task SendGlobalMessage(string message) {
        await _connection.InvokeAsync("SendGlobalMessage", message);
    }

    public async Task SendGuildMessage(int guildId, string message) {
        await _connection.InvokeAsync("SendGuildMessage", guildId, message);
    }

    public async Task SendWhisper(string targetUsername, string message) {
        await _connection.InvokeAsync("SendWhisper", targetUsername, message);
    }
}
```

**데미지 계산**:
```csharp
// 캐릭터의 Attack 기반 데미지
public long CalculateRaidDamage(Character character) {
    // 기본 데미지 + 장비 + 펫 + 길드 버프
    long damage = character.Attack;

    // 크리티컬 (20% 확률로 2배)
    if (Random.value < 0.2f) {
        damage *= 2;
    }

    return damage;
}
```

```csharp
public class DailyResetService : BackgroundService {
    protected override async Task ExecuteAsync(CancellationToken stoppingToken) {
        while (!stoppingToken.IsCancellationRequested) {
            var now = DateTime.Now;
            var nextMidnight = now.Date.AddDays(1);
            var delay = nextMidnight - now;

            await Task.Delay(delay, stoppingToken);

            // 일일 미션 리셋
            await _dailyMissionService.ResetDailyMissions();

            // 출석 연속일 체크 (어제 출석 안한 사람은 ConsecutiveDays = 0)
```

**확률 시스템**:
```csharp
public List<Item> PullGacha(int userId, int bannerId, int count) {
    var character = _context.Characters.Find(userId);
    var results = new List<Item>();

    for (int i = 0; i < count; i++) {
        // 천장 시스템 (90회)
        if (character.GachaPityCounter >= 90) {
            results.Add(GetRandomItem(Rarity.Legendary));
            character.GachaPityCounter = 0;
            continue;
        }

        float roll = Random.value;
        Rarity rarity;

        if (roll < 0.006f) {        // 0.6% 전설
            rarity = Rarity.Legendary;
            character.GachaPityCounter = 0;
        } else if (roll < 0.056f) { // 5.0% 에픽
            rarity = Rarity.Epic;
        } else if (roll < 0.256f) { // 20.0% 레어
            rarity = Rarity.Rare;
        } else {                    // 74.4% 일반
            rarity = Rarity.Common;
        }

        character.GachaPityCounter++;
        results.Add(GetRandomItem(rarity));
    }

    // 10연차 보장: 레어 이상 1개 보장
    if (count == 10) {
        var hasRareOrAbove = results.Any(r => r.Rarity >= Rarity.Rare);
        if (!hasRareOrAbove) {
            results[9] = GetRandomItem(Rarity.Rare);
        }
    }

    return results;
}
```
